无重复字符的最长子串

  • description: 给定一个字符串 s ，请你找出其中不含有重复字符的 最长 子串 的长度。
int lengthOfLongestSubstring(char* s) {
    int res = 0;
    int sLen = strlen(s);
    int left = 0;
    int lastPos[128];
    for (int i = 0; i < 128; i++) {
        lastPos[i] = -1;
    }

    for (int right = 0; right < sLen; right++) {
        char c_in = s[right];
        if (lastPos[c_in] >= left) {
            left = lastPos[c_in] + 1;
        }
        lastPos[c_in] = right;
        int curLen = right - left + 1;
        res = curLen > res ? curLen : res;
    }
    return res;
}

找到字符串中所有字母异位词

  • description: 给定两个字符串 s 和 p，找到 s 中所有 p 的 异位词 的子串，返回这些子串的起始索引。不考虑答案输出的顺序。
int *findAnagrams(char *s, char *p, int *returnSize) {
    int sLen = strlen(s);
    int pLen = strlen(p);
    int *res = (int *) malloc(sizeof(int) * sLen);
    *returnSize = 0;
    int hashP[26] = {0};
    int hashWindow[26] = {0};
    int left = 0;
    int right = 0;
    int count = 0;

    for (int i = 0; i < pLen; i++) {
        hashP[p[i] - 'a']++;
    }

    while (right < sLen) {
        char c_in = s[right++];

        if (hashWindow[c_in - 'a'] < hashP[c_in - 'a']) {
            count++;
        }
        hashWindow[c_in - 'a']++;

        if (right - left > pLen) {
            char c_out = s[left++];
            if (hashWindow[c_out - 'a'] <= hashP[c_out - 'a']) {
                count--;
            }
            hashWindow[c_out - 'a']--;
        }
        if (count == pLen) {
            res[(*returnSize)++] = left;
        }
    }
    return res;
}
