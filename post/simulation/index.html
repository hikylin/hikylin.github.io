柜台办理

某运营商提供num个柜台供业务办理，客户到达后先进入等候区排队，然后按规则到柜台办理。现有一批客户信息按到达先后顺序记录与customers,customers[i]=[arriveTime, userType, processTime]表示客户i的类型为userType(1:普通用户， 2：VIP客户)，在arriveTime时刻到达(arriveTime递增))，办理业务时长为processTime。 柜台办理规则如下：

  • 当客户到达时，先排队，VIP优先；普通客户排在队尾，VIP客户插到队中首个普通用户前面，已有VIP客户的后面。
  • 一旦柜面出现空闲，则队首客户到任意空闲柜台办理。 最后按输入顺序返回每个客户处理完业务的时刻。

输入

第一个参数为num, 1<= num<=10 第二个参数为customers,1<=customers.length<=1000, 0<=arriveTime<=10^ 99, 1<=userType<=2, 1<=processTime<=10^ 6 用例保证arriveTime递增。

输出

每个客户办理完业务的时刻列表

代码示例

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int arriveTime;
    int userType; // 1: 普通, 2: VIP
    int processTime;
} Customer;

/**
 * @brief 模拟银行排队系统，计算每个客户的业务完成时间
 * @param num 柜台数量
 * @param customers 客户数组（已按到达时间排序）
 * @param customersSize 客户总数
 * @param returnSize 输出参数，返回结果数组大小
 * @return 动态分配的整数数组，对应每个客户的完成时间
 */
static int *GetUserFinishedTime(int num, const Customer *customers, int customersSize, int *returnSize) {
    // 1. 边界检查
    if (customersSize == 0 || num <= 0) {
        *returnSize = 0;
        return NULL;
    }

    *returnSize = customersSize;
    int *result = (int *)malloc(sizeof(int) * customersSize);
    if (!result) return NULL;

    // 辅助数组
    int *processed = (int *)calloc(customersSize, sizeof(int));      // 标记客户是否已办理
    long long *counterFreeTime = (long long *)calloc(num, sizeof(long long)); // 记录每个柜台的空闲时刻

    int processedCount = 0;
    // 初始化系统时间：从第一个客户到达时刻开始
    long long currentTime = customers[0].arriveTime;

    // 2. 主循环：直到所有客户办理完毕
    while (processedCount < customersSize) {
        int bestCustomerIdx = -1; // 本轮选中的客户
        int bestCounterIdx = -1;  // 分配给该客户的柜台

        // --- 阶段 A: 尝试在当前时刻匹配“空闲柜台”与“队首客户” ---
        for (int i = 0; i < num; i++) {
            // 只有柜台空闲（空闲时刻 <= 当前时间）才考虑分配
            if (counterFreeTime[i] <= currentTime) {
                int vipIdx = -1;
                int normalIdx = -1;

                // 遍历寻找“已到达且未处理”的客户
                // 策略：优先找 VIP，若找不到 VIP 再找普通人
                for (int j = 0; j < customersSize; j++) {
                    if (!processed[j] && customers[j].arriveTime <= currentTime) {
                        if (customers[j].userType == 2) {
                            vipIdx = j;
                            break; // 找到第一个 VIP 即可，VIP 优先级最高
                        }
                        if (normalIdx == -1) {
                            normalIdx = j; // 记录第一个普通人
                        }
                    }
                }

                // 决策：有 VIP 选 VIP，无 VIP 选普通人
                if (vipIdx != -1) {
                    bestCustomerIdx = vipIdx;
                    bestCounterIdx = i;
                    break; // 命中最高优先级，立即跳出柜台循环去处理
                } else if (normalIdx != -1) {
                    bestCustomerIdx = normalIdx;
                    bestCounterIdx = i;
                    break; // 命中普通人，立即跳出（任意空闲柜台均可）
                }
            }
        }

        // --- 阶段 B: 执行办理 或 时间跳跃 ---
        if (bestCustomerIdx != -1) {
            // 【情况 1: 成功匹配】立即办理业务
            long long startTime = (counterFreeTime[bestCounterIdx] > customers[bestCustomerIdx].arriveTime) 
                                  ? counterFreeTime[bestCounterIdx] 
                                  : customers[bestCustomerIdx].arriveTime;
            
            long long finishTime = startTime + customers[bestCustomerIdx].processTime;
            
            // 更新状态
            counterFreeTime[bestCounterIdx] = finishTime;
            result[bestCustomerIdx] = (int)finishTime;
            processed[bestCustomerIdx] = 1;
            processedCount++;
            
            // 注意：不增加 currentTime，因为同一时刻可能有多个柜台同时空闲，需继续循环处理
        } else {
            // 【情况 2: 无法匹配】需要推进时间
            // 原因可能是：所有柜台都忙，或者柜台有空但还没人到达
            
            // 2.1 计算下一个关键时间点：最早的一个柜台变空闲的时间
            // 既然能进到这里，说明要么全忙，要么没人。nextCounterFree 必然存在且有效。
            long long nextCounterFree = counterFreeTime[0];
            for (int i = 1; i < num; i++) {
                if (counterFreeTime[i] < nextCounterFree) {
                    nextCounterFree = counterFreeTime[i];
                }
            }

            // 2.2 计算下一个关键时间点：下一个未处理客户的到达时间
            long long nextArrival = -1;
            for (int i = 0; i < customersSize; i++) {
                if (!processed[i]) {
                    nextArrival = customers[i].arriveTime;
                    break; // 输入有序，第一个即为最早
                }
            }

            // 2.3 决定跳转到哪个时间
            long long jumpTime;
            if (nextArrival == -1) {
                // 所有人都已到达（在排队或办理中），只能等柜台空闲
                jumpTime = nextCounterFree;
            } else {
                // 还有人没到，取“最早柜台空闲”和“下一个人到达”的较小值
                jumpTime = (nextCounterFree < nextArrival) ? nextCounterFree : nextArrival;
            }

            // 2.4 更新系统时间（防御性：确保时间严格向前推进，防止死循环）
            if (jumpTime > currentTime) {
                currentTime = jumpTime;
            } else {
                currentTime++; 
            }
        }
    }

    free(processed);
    free(counterFreeTime);
    return result;
}
